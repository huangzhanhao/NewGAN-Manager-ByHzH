```python
import os
import shutil
import sys
import tempfile

# 让测试能直接 import core 模块（与 CI 的 unittest discover -s src/tests 兼容）
sys.path.insert(0, os.path.join(os.path.dirname(os.path.abspath(__file__)), "..", "newganmanager"))

TESTS_DIR = os.path.dirname(os.path.abspath(__file__))
TESTING_DATA = os.path.join(TESTS_DIR, "testing_data")


def copy_testing_data():
    """把 testing_data 完整复制到临时目录，返回副本路径；调用方负责清理"""
    tmp = tempfile.mkdtemp(prefix="newgan_test_")
    dst = os.path.join(tmp, "testing_data")
    shutil.copytree(TESTING_DATA, dst)
    return dst


def make_facepack(root, ethnicities):
    """创建临时头像包目录：每个种族目录下生成 N 张假图片文件

    Args:
        root: 头像包根目录
        ethnicities: {目录名: 图片数量}，图片名为 e0.png, e1.png, ...
    Returns:
        root（调用方负责清理其父目录）
    """
    for eth, count in ethnicities.items():
        eth_dir = os.path.join(root, eth)
        os.makedirs(eth_dir, exist_ok=True)
        for i in range(count):
            with open(os.path.join(eth_dir, f"e{i}.png"), "wb") as fp:
                fp.write(b"\x89PNG\r\n\x1a\n")  # PNG magic bytes，仅占位
    return root
```

```python
import json
import os


class ConfigManager:
    def __init__(self):
        pass

    def load_config(self, path):
        try:
            with open(path, "r", encoding="utf-8") as fp:
                return json.load(fp)
        except FileNotFoundError:
            raise FileNotFoundError(f"Config file not found: {path}")
        except json.JSONDecodeError as e:
            backup = self._backup_corrupt_file(path)
            message = f"Config file is corrupted and could not be parsed: {path}"
            if backup:
                message += f" (backed up to {backup})"
            raise ValueError(message) from e

    def save_config(self, path, data):
        """原子写入：先写临时文件再 os.replace，避免进程中途崩溃留下截断的 JSON"""
        tmp_path = f"{path}.tmp"
        try:
            with open(tmp_path, "w", encoding="utf-8") as fp:
                json.dump(data, fp, ensure_ascii=False, indent=2)
        except FileNotFoundError:
            self._remove_file(tmp_path)
            raise FileNotFoundError(f"Config path not found: {path}")
        except PermissionError:
            self._remove_file(tmp_path)
            raise PermissionError(f"Permission denied when saving config file: {path}")
        except TypeError:
            self._remove_file(tmp_path)
            raise TypeError(f"Config data is not JSON serializable: {path}")
        os.replace(tmp_path, path)

    @staticmethod
    def _remove_file(path):
        try:
            os.remove(path)
        except OSError:
            pass

    @staticmethod
    def _backup_corrupt_file(path):
        """把损坏的配置文件改名为 <path>.corrupt，返回备份路径；失败返回 None"""
        backup_path = f"{path}.corrupt"
        try:
            os.replace(path, backup_path)
            return backup_path
        except OSError:
            return None

    def get_latest_prf(self, path):
        """从配置中获取最新的活动配置文件名称"""
        try:
            cfg = self.load_config(path)
            for k, v in cfg["Profile"].items():
                if v:
                    return k
            return None
        except Exception:
            return None
```

```python
        self.assertIn("r-1915576430", uids)
        for uid in uids:
```

```python
import glob
import os

import unittest

from testutils import copy_testing_data
from newganmanager.core.XmlParser import XmlParser


class TestXmlParser(unittest.TestCase):
    def setUp(self):
        self.data_dir = copy_testing_data()
        self.parser = XmlParser()
        self.img_dir = os.path.join(self.data_dir, "facepack")
        os.makedirs(self.img_dir, exist_ok=True)

    def tearDown(self):
        shutil.rmtree(self.data_dir, ignore_errors=True)

    def _config_path(self):
        return os.path.join(self.img_dir, "config.xml")

    def _write_config(self, lines):
        with open(self._config_path(), "w", encoding="UTF-8") as fp:
            fp.write("<record>\n<list id=\"maps\">\n")
            for uid, eth, img in lines:
                fp.write(f'                <record from="{eth}/{img}" to="graphics/pictures/person/{uid}/portrait"/>\n')
            fp.write("</list>\n</record>\n")

    def test_parse_xml(self):
        self._write_config([("r-1234", "African", "a1"), ("5678", "Caucasian", "c1")])
        data = self.parser.parse_xml(self._config_path())
        self.assertEqual(data["r-1234"], {"uid": "r-1234", "ethnicity": "African", "image": "a1"})
        self.assertEqual(data["5678"], {"uid": "5678", "ethnicity": "Caucasian", "image": "c1"})

    def test_parse_xml_missing_file_raises(self):
        with self.assertRaises(FileNotFoundError):
            self.parser.parse_xml(self._config_path())

    def test_get_imgpath_from_uid(self):
        self._write_config([("r-1234", "African", "a1"), ("5678", "Caucasian", "c1")])
        self.assertEqual(self.parser.get_imgpath_from_uid(self._config_path(), "r-1234"), "African/a1")
        self.assertIsNone(self.parser.get_imgpath_from_uid(self._config_path(), "9999"))

    def test_write_xml(self):
        mapping = [["r-1234", "African", "a1"], ["r-5678", "Caucasian", "c1"]]
        # root_dir 用 testing_data 副本（内含 .config/config_template）
        result = self.parser.write_xml(mapping, self.img_dir, self.data_dir, self.parser.logger)
        self.assertEqual(len(result), 2)
        with open(self._config_path(), "r", encoding="UTF-8") as fp:
            content = fp.read()
        self.assertIn('<record from="African/a1" to="graphics/pictures/person/r-1234/portrait"/>', content)
        self.assertIn('<boolean id="preload" value="false"/>', content)  # 模板头尾保留
        # 回读一致
        data = self.parser.parse_xml(self._config_path())
        self.assertEqual(data["r-1234"]["image"], "a1")

    def test_write_xml_creates_backup(self):
        self._write_config([("r-1", "African", "old")])
        self.parser.write_xml([["r-2", "African", "new"]], self.img_dir, self.data_dir, self.parser.logger, save_backup=True)
        backups = glob.glob(os.path.join(self.img_dir, "config备份_*.xml"))
        self.assertEqual(len(backups), 1)

    def test_backup_rotation_max_10(self):
        """备份最多保留 10 份，最旧的被清理"""
        for i in range(12):
            self._write_config([("r-1", "African", f"v{i}")])
            self.parser._save_backup_config_xml(self._config_path(), self.img_dir, self.parser.logger)
        backups = glob.glob(os.path.join(self.img_dir, "config备份_*.xml"))
        self.assertEqual(len(backups), 10)

        self.parser.single_replacement_in_xml(
            ["r-9999", "Asian", "x9"], self.img_dir, self.parser.logger, save_backup=False)
        data = self.parser.parse_xml(self._config_path())
        self.assertEqual(len(data), 1)

    unittest.main()
```

```python
    def _get_image_pool(self, ethnicity, nationality):
        """获取优先级图片池：[(目录名, 图片集合), ...]，国籍同名目录 > 种族目录 > 全部目录"""
        pools = []
        if nationality in self.faces_map and self.faces_map[nationality]:
            pools.append((nationality, self.faces_map[nationality]))
        if ethnicity in self.faces_map and self.faces_map[ethnicity]:
            pools.append((ethnicity, self.faces_map[ethnicity]))
        if not pools and self.faces_map:
            # 国籍与种族目录都不在头像包内时，回退到全部目录
            pools = list(self.faces_map.items())
        return pools
```

`pick_image_from_pools` 返回 `(目录名, 图片名)`；删除 `pick_image`（仓库内无调用方）：

```python
    def pick_image_from_pools(self, pools, duplicates):
        """从多个 (目录名, 图片集合) 池中选一张图片，返回 (目录名, 图片名) 或 None"""
        for name, pool in pools:
```

```python
import hashlib
import logging
import os
import shutil

from .ConfigManager import ConfigManager


class ProfileManager(ConfigManager):
    def __init__(self, name, root_dir, data_dir=None):
        """
        Args:
            name (str): 当前 Profile 名称
            root_dir (str): 应用资源目录（.config 下的只读配置随程序分发）
            data_dir (str): 用户数据目录（.user 运行时数据，来自 paths.data；
                            缺省时与 root_dir 相同，兼容旧布局/测试）
        """
        super().__init__()
        self.root_dir = root_dir
        self.data_dir = data_dir if data_dir is not None else root_dir
        self.config = self.load_config(self.user_path("cfg.json"))
        self.prf_cfg = self.load_config(self.user_path(name + ".json"))
        self.eth_cfg = self.load_config(os.path.join(root_dir, ".config", "eth_cfg.json"))
        self.cur_prf = name
        self.logger = logging.getLogger("NewGAN App")
        self._normalize_img_dirs()
        self._migrate_legacy_snapshot()

    def user_path(self, filename):
        """返回用户数据目录（data_dir/.user）下指定文件的路径"""
        return os.path.join(self.data_dir, ".user", filename)

    # ---- 多头像包目录管理 ----

    def _normalize_img_dirs(self):
        """把头像包目录字段规范为 img_dirs 列表 + 激活目录 img_dir（兼容旧版单目录字段）"""
        img_dirs = self.prf_cfg.get("img_dirs")
        if not isinstance(img_dirs, list):
            img_dirs = []
        img_dirs = [self._normalize_dir(d) for d in img_dirs if d]
        active = self._normalize_dir(self.prf_cfg.get("img_dir", ""))
        if active and active not in img_dirs:
            img_dirs.insert(0, active)
        self.prf_cfg["img_dirs"] = img_dirs
        self.prf_cfg["img_dir"] = active if active in img_dirs else (img_dirs[0] if img_dirs else "")

    @staticmethod
    def _normalize_dir(path):
        """统一目录字符串：去掉尾部斜杠，避免后续拼接出混合分隔符路径"""
        return path.rstrip("/\\") if isinstance(path, str) else ""

    def add_img_dir(self, img_dir):
        """登记一个新的头像包目录并设为激活；已存在时仅切换激活"""
        img_dir = self._normalize_dir(img_dir)
        if not img_dir:
            return
        if img_dir not in self.prf_cfg["img_dirs"]:
            self.prf_cfg["img_dirs"].append(img_dir)
        self.prf_cfg["img_dir"] = img_dir

    def remove_img_dir(self, img_dir):
        """从头像包目录列表移除并清理其快照；移除激活目录时激活切到剩余第一个"""
        img_dir = self._normalize_dir(img_dir)
        if img_dir in self.prf_cfg["img_dirs"]:
            self.prf_cfg["img_dirs"].remove(img_dir)
        if self.prf_cfg["img_dir"] == img_dir:
            self.prf_cfg["img_dir"] = self.prf_cfg["img_dirs"][0] if self.prf_cfg["img_dirs"] else ""
        snap = self._snapshot_path(self.cur_prf, img_dir)
        if os.path.isfile(snap):
            try:
                os.remove(snap)
            except OSError as e:
                self.logger.error(f"Error removing snapshot for {img_dir}: {e}")

    def set_active_img_dir(self, img_dir):
        img_dir = self._normalize_dir(img_dir)
        if img_dir and img_dir in self.prf_cfg["img_dirs"]:
            self.prf_cfg["img_dir"] = img_dir

    def _snapshot_path(self, profile, img_dir):
        """某个 (Profile, 头像包目录) 组合的 config.xml 快照路径"""
        p = hashlib.md5(profile.encode("utf-8")).hexdigest()[:12]
        d = hashlib.md5(img_dir.encode("utf-8")).hexdigest()[:12]
        return self.user_path(os.path.join("snapshots", f"{p}_{d}.xml"))

    def _migrate_legacy_snapshot(self):
        """旧版单快照 .user/<profile>.xml → 多目录快照 .user/snapshots/"""
        legacy = self.user_path(self.cur_prf + ".xml")
        if not os.path.isfile(legacy):
            return
        active = self.prf_cfg["img_dir"]
        if active:
            new_path = self._snapshot_path(self.cur_prf, active)
            os.makedirs(os.path.dirname(new_path), exist_ok=True)
            try:
                shutil.copyfile(legacy, new_path)
            except OSError as e:
                self.logger.error(f"Error migrating legacy snapshot: {e}")
                return
        try:
            os.remove(legacy)
            self.logger.info(f"Migrated legacy snapshot for profile '{self.cur_prf}'")
        except OSError as e:
            self.logger.error(f"Error removing legacy snapshot {legacy}: {e}")

    # ---- Profile 生命周期 ----

    def delete_profile(self, name):
        if name == "No Profile":
            self.logger.warning("Can't delete no profile")
            return False
        del self.config["Profile"][name]
        try:
            os.remove(self.user_path(name + ".json"))
            self._remove_snapshots(name)
        except OSError as e:
            self.logger.error(f"Error deleting profile files: {e}")
        self.save_config(self.user_path("cfg.json"), self.config)
        self.load_profile("No Profile")
        self.logger.info(f"Delete profile: {name}")
        return True

    def _remove_snapshots(self, name):
        prefix = hashlib.md5(name.encode("utf-8")).hexdigest()[:12] + "_"
        snap_dir = self.user_path("snapshots")
        if not os.path.isdir(snap_dir):
            return
        for fname in os.listdir(snap_dir):
            if fname.startswith(prefix):
                try:
                    os.remove(os.path.join(snap_dir, fname))
                except OSError as e:
                    self.logger.error(f"Error removing snapshot {fname}: {e}")

    def create_profile(self, name):
        self.config["Profile"][name] = True
        self.save_config(self.user_path("cfg.json"), self.config)
        self.save_config(
            self.user_path(name + ".json"),
            {"img_dirs": [], "img_dir": "", "rtf": ""},
        )
        self.load_profile(name)
        self.logger.info(f"Create new profile: {name}")

    def load_profile(self, name):
        deact_cfg = self.prf_cfg
        self.prf_cfg = self.load_config(self.user_path(name + ".json"))
        self._normalize_img_dirs()
        # 交换两个 Profile 各自头像包目录下的 config.xml（先备份旧目录，再恢复新目录）
        self.swap_xml(self.cur_prf, name, deact_cfg, self.prf_cfg)
        # 遍历所有配置文件，将当前配置文件设为激活状态，其他设为非激活状态
        for key in self.config["Profile"]:
            self.config["Profile"][key] = key == name
        self.cur_prf = name
        self.save_config(self.user_path(name + ".json"), self.prf_cfg)

    def swap_xml(self, deact_name, act_name, deact_cfg, act_cfg):
        """在停用与激活 Profile 之间交换各头像包目录的 config.xml

        两阶段执行：先把停用 Profile 各目录下的 config.xml 备份为快照，
        再把激活 Profile 的快照写回其各目录——保证两个 Profile 共用同一
        头像包目录时不会被激活方的快照覆盖掉待备份内容。
        """
        try:
            # 阶段1：备份停用 Profile 各目录的 config.xml
            for img_dir in deact_cfg.get("img_dirs", []):
                config_file = os.path.join(img_dir, "config.xml")
                if os.path.isfile(config_file):
                    snap = self._snapshot_path(deact_name, img_dir)
                    os.makedirs(os.path.dirname(snap), exist_ok=True)
                    shutil.copyfile(config_file, snap)
            # 阶段2：把激活 Profile 的快照写回其各目录
            for img_dir in act_cfg.get("img_dirs", []):
                snap = self._snapshot_path(act_name, img_dir)
                if os.path.isfile(snap):
                    os.makedirs(img_dir, exist_ok=True)
                    shutil.copyfile(snap, os.path.join(img_dir, "config.xml"))
        except (IOError, OSError) as e:
            self.logger.error(f"Error swapping XML files: {e}")

    def get_ethnic(self, nation):
        return self.eth_cfg["Ethnics"].get(nation, None)
```

```python
    def _lookup_preview_data(self, uid):
        """在工作线程中完成全部查找：返回 (ethnicity, image_name, image_file, player)"""
        ethnicity = None
        image_name = None
        # 1) 最近一次映射结果
        mapping_data = self.main_tab.replace_service.mapping_data
        if mapping_data:
            mapped = next((p for p in mapping_data if p and len(p) > 0 and (p[0] == uid or p[0] == "r-" + uid)), None)
            if mapped:
                ethnicity, image_name = mapped[1], mapped[2]
        # 2) 未命中再查 config.xml
        if ethnicity is None or image_name is None:
            self.logger.info(f"UID {uid} not found in mapping_data, checking XML file")
            img_path = XmlParser().get_imgpath_from_uid(
                os.path.join(self.app.profile_manager.prf_cfg["img_dir"], "config.xml"), uid)
            if img_path:
                parts = img_path.split("/")
                if len(parts) >= 2:
                    ethnicity, image_name = parts[0], parts[1]
                else:
                    self.logger.warning(f"Invalid image path format for UID {uid}: {img_path}")
            else:
                self.logger.warning(f"UID {uid} not found in config.xml file")
        # 3) 定位图片文件
        image_file = None
        if ethnicity is not None and image_name is not None:
            img_base = os.path.join(self.app.profile_manager.prf_cfg['img_dir'], ethnicity, image_name)
            for ext in ('.png', '.jpg', '.jpeg'):
                if os.path.isfile(img_base + ext):
                    image_file = img_base + ext
                    break
            if image_file is None:
                self.logger.warning(f"Image file not found for UID {uid}")
        # 4) 球员详情
        player = None
        rtf_data = self.main_tab.replace_service.rtf_data
        if rtf_data:
            player = next((p for p in rtf_data if p and len(p) > 0 and (p[0] == uid or p[0] == "r-" + uid)), None)
            if player is None:
                self.logger.warning(f"Player details not found in RTF data for UID: {uid}")
        else:
            self.logger.warning("No RTF data available for previewing player details")
        return ethnicity, image_name, image_file, player

        ethnicity, image_name, image_file, player = result
        if ethnicity is not None and image_name is not None:
            self.img_path.text = f"{ethnicity}\\{image_name}"
            if image_file:
                self.rep_img.image = toga.Image(image_file)
```

```python
class UIHandler(logging.Handler):
    """Handler that forwards log records to a UI queue."""

    def __init__(self, ui_queue: asyncio.Queue):
        super().__init__()
        self.ui_queue = ui_queue
        self.loop = None  # 应用事件循环启动后由 attach_ui_handler 绑定

    def attach_loop(self, loop):
        self.loop = loop

    def emit(self, record: logging.LogRecord):
        if self.loop is None:
            return  # UI 循环未启动，日志只落文件
        try:
            asyncio.run_coroutine_threadsafe(self.ui_queue.put(record), self.loop)
        except Exception:
            self.handleError(record)
```
